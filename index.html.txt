    <script>
        const tg = window.Telegram.WebApp;
        tg.expand();

        function numberToWordsUZ(n) {
            if (n === 0) return "nol";
            const birliklar = ["", "bir", "ikki", "uch", "to'rt", "besh", "olti", "yetti", "sakkiz", "to'qqiz"];
            const onliklar = ["", "o'n", "yigirma", "o'ttiz", "qirq", "ellik", "oltmish", "yetmish", "sakson", "to'qson"];
            let res = "";
            if (n >= 1000000) {
                let m = Math.floor(n / 1000000);
                res += birliklar[m] + " million ";
                n %= 1000000;
            }
            if (n >= 1000) {
                let t = Math.floor(n / 1000);
                if (t >= 100) { res += birliklar[Math.floor(t / 100)] + " yuz "; t %= 100; }
                if (t >= 10) { res += onliklar[Math.floor(t / 10)] + " "; }
                if (t % 10 > 0) { res += birliklar[t % 10] + " "; }
                res += "ming ";
                n %= 1000;
            }
            if (n >= 100) { res += birliklar[Math.floor(n / 100)] + " yuz "; n %= 100; }
            if (n >= 10) { res += onliklar[Math.floor(n / 10)] + " "; n %= 10; }
            if (n > 0) { res += birliklar[n] + " "; }
            return res.trim();
        }

        function calculate() {
            let val = document.getElementById('amount').value;
            if(!val) {
                alert("Iltimos summani kiriting!");
                return;
            }
            let amount = parseFloat(val);
            let usdSell = 11900;
            let eurSell = 15000;

            let usd = amount / usdSell;
            let eur = amount / eurSell;
            let uzWords = numberToWordsUZ(Math.round(amount));

            let text = `🇺🇿 Shartnoma uchun (O'zbekcha):\n` +
                       `${amount.toLocaleString()} UZS (${uzWords} so'm)\n` +
                       `Ekvivalenti: $${usd.toFixed(2)} / €${eur.toFixed(2)}\n\n` +
                       `🇷🇺 Для договора (Русский):\n` +
                       `${amount.toLocaleString()} сум\n` +
                       `Эквивалент: $${usd.toFixed(2)} / €${eur.toFixed(2)}`;

            document.getElementById('result').style.display = 'block';
            document.getElementById('result').innerText = text;
            document.getElementById('copyBtn').style.display = 'block';
        }

        function copyText() {
            let textToCopy = document.getElementById('result').innerText;
            navigator.clipboard.writeText(textToCopy);
            tg.showAlert("Matn nusxalandi!");
        }
